\begin{theorem}[Fundamental Theorem of Arithmetic, \cite{burton2011}]
    Every integer $n > 1$ can be expressed as a product of primes. Moreover,
this representation is \emph{unique} up to the order of the factors. That
is, there exist primes $p_1 \leq p_2 \leq \cdots \leq p_k$ such that
\[
    n \;=\; p_1^{e_1} \cdot p_2^{e_2} \cdots p_k^{e_k},
\]
where each $e_i \in \bbn$, and this factorization is unique
up to reordering of the prime factors.
\end{theorem}
